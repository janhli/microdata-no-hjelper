// S_TOPOGRAFIICD10 -> kreftform som i Cancer in Norway (48 kreftformer, tabell 2.3 i Cancer in Norway 2025, gjengitt på fhi.no).

// C-koder som ikke er kreftform i rapporten, står uendret: C46 C77 C78 C79 C97 (C46 Kaposi er ført under organet, C77-C79 er spredning, C97 flere svulster)
generate kreftform = substr(topo, 1, 3)
replace kreftform = substr(topo, 1, 4) if inlist(substr(topo, 1, 4), 'C000', 'C001', 'C002', 'C003', 'C004', 'C005', 'C006', 'C008', 'C009', 'C051', 'C052', 'C570', 'C571', 'C572', 'C573', 'C574', 'C577', 'C578', 'C579', 'D352', 'D353', 'D354', 'D443', 'D444', 'D445')
