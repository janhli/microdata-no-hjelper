// Spør om versjonen som er tilgjengelig i analysemiljøet (kan ligge en bak metadatasidene)
require no.ssb.fdb:57 as fdb

// --- 1. Populasjon ---
create-dataset personer
// Første import bestemmer populasjonen. values() filtrerer allerede ved import.
import fdb/BEFOLKNING_KOMMNR_FORMELL 2025-01-01 as kommune
generate fylke = substr(kommune, 1, 2)

// --- 2. Kjennetegn ---
import fdb/BEFOLKNING_KJOENN as kjonn
import fdb/BEFOLKNING_FOEDSELS_AAR_MND as faarmnd

// --- 3. Avledede variabler ---
generate alder = 2025 - 1 - int(faarmnd/100)
generate aldgr = '1'
replace aldgr = '2' if alder >= 20
replace aldgr = '3' if alder >= 67
replace aldgr = '9' if sysmiss(alder)
define-labels aldgrtxt '1' '0–19 år' '2' '20–66 år' '3' '67 år og over' '9' 'Ukjent'
assign-labels aldgr aldgrtxt
