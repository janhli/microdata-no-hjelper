// =====================================================================
// NEET: unge 15–29 år utenfor arbeid, utdanning og arbeidsmarkedstiltak
// Status:    testet (no.ssb.fdb:60, 2026-11)
// Variabler: ARBSTATUS_PERS_NEET (status, 4. kvartal; 2008-11-16 til 2024-11-16), '1' = NEET, '0' = ikke
//            https://microdata.no/discovery/variable/no.ssb.fdb/60/ARBSTATUS_PERS_NEET
//            ARBSTATUS_PERS_STATUS8 (prioritert arbeidsstyrkestatus, samme dato)
//            https://microdata.no/discovery/variable/no.ssb.fdb/60/ARBSTATUS_PERS_STATUS8
//            ARBSTATUS_PERS_ALDER (alder på måletidspunktet)
//            https://microdata.no/discovery/variable/no.ssb.fdb/60/ARBSTATUS_PERS_ALDER
//            ARBSTATUS_PERS_KOMM_NR (bostedskommune, datoene er den 30., ikke den 16.)
//            https://microdata.no/discovery/variable/no.ssb.fdb/60/ARBSTATUS_PERS_KOMM_NR
//            BEFOLKNING_KJOENN, BEFOLKNING_INVKAT (faste)
//            https://microdata.no/discovery/variable/no.ssb.fdb/60/BEFOLKNING_KJOENN
//            https://microdata.no/discovery/variable/no.ssb.fdb/60/BEFOLKNING_INVKAT
//            NUDB_BU (høyeste fullførte utdanning, siste dato 2024-08-01)
//            https://microdata.no/discovery/variable/no.ssb.fdb/60/NUDB_BU
// Lager:     neet      – '1' NEET, '0' i arbeid, utdanning eller tiltak (SSBs egen variabel)
//            neetstat  – for NEET: '1' registrert ledig, '2' AAP/uføretrygd, '3' ukjent status, '4' andre
//                        ytelser. '0' for dem som ikke er NEET
//            aldgr     – '1' 15–19, '2' 20–24, '3' 25–29 år
//            fylke     – bostedsfylke (2024-inndelingen) med navn
//            innv      – '1' innvandrer, '2' resten (som SSB tabell 12423)
//            utdniv    – '1' grunnskole, '2' videregående (inkl. fagskole), '3' universitet/høgskole,
//                        '9' uoppgitt/ingen (som SSB tabell 12423)
// Definisjon (SSB): NEET er bosatte 15–29 år som ikke er sysselsatt, under ordinær utdanning eller på
//            arbeidsmarkedstiltak. Måletidspunkt 3. uke i november. Testet med hovedstatus (de fire første
//            sifrene i STATUS8): ikke NEET er sysselsatte (0808, 0816, 0832, 1608), ordinær utdanning (4008)
//            og tiltak (3208, 3216 tiltakspenger, 3240 introduksjonsprogram, 4816). Alle andre er NEET, også
//            dagpenger (3224), sykepenger (4808), sosialhjelp, kontantstøtte, overgangsstønad og ukjent (7208).
// Kontroll:  SSB tabell 13556 (NEET etter region, kjønn og alder) og 12423 (etter innvandringskategori og
//            utdanningsnivå), 2024, hentet 2026-11-23. Tallene står i tekstblokkene under. Kjørt 2026-11-23:
//            102 977 NEET av 1 044 100 (SSB 102 976 av 1 044 101). Ledige 11 659 (11 656), AAP/uføre 27 131
//            (27 134), ukjent og andre 64 185 (64 186). Alder, kjønn, alle 15 fylker og innvandrere/resten
//            innen 0,1 prosentpoeng. Ukjent status er 54 420 (53 % av NEET). Utdanning avviker, se under.
// Fallgruver:
//   - Variabelen dekker bare til 2024-11-16 (versjon 60). Det finnes ikke NEET for 2025 i microdata.no,
//     selv om SSB har publisert foreløpige tall.
//   - Brudd mellom 2014 og 2015 (a-ordningen). Sammenlign ikke nivået før og etter.
//   - Importer NEET først, i et tomt datasett. Da blir populasjonen den samme som hos SSB (i underkant av
//     50 færre i microdata.no). Starter du med bosatte fra BEFOLKNING_KOMMNR_FORMELL, får du en annen
//     populasjon, med manglende NEET for dem som ikke var 15–29 i november.
//   - Bruk alder og bosted fra ARBSTATUS-variablene, ikke fødselsår og BEFOLKNING_KOMMNR_FORMELL, så
//     aldersgruppene og fylkene blir de samme som i SSBs tabeller.
//   - Andeler: bruk tabeller med rowpct, ikke summarize(x) mean. Ledige og AAP/uføre er under 1 % i flere
//     grupper (15–19 år: 0,4 %), og sjeldne 0/1-variabler blir 0 i gjennomsnitt (se fallgruver.md, «Analyse»).
//   - ARBSTATUS_PERS_UTENFOR er samme definisjon for alle 15 år og eldre, og regner også AFP og alderspensjon
//     som «utenfor».
//   - Registerdata gir en større NEET-gruppe enn Arbeidskraftundersøkelsen (AKU). Internasjonale
//     sammenligninger bruker AKU. Halvparten av NEET-gruppen har ukjent status: de finnes ikke i registrene
//     over arbeid, utdanning og ytelser (utvandret uten å melde fra, svart arbeid, utdanning i utlandet mv.).
//   - Utdanningsnivået treffer ikke SSB 12423. NEET-andelen var 14,0 % for grunnskole (SSB 12,9), 6,8 for
//     videregående (7,5), 5,6 for universitet/høgskole (6,9) og 16,1 for uoppgitt (54,4). Antall personer:
//     uoppgitt 104 911 her mot 7 417 hos SSB, grunnskole 351 190 mot 425 984, videregående 324 039 mot
//     333 069, universitet/høgskole 263 955 mot 277 630. SSB har altså fordelt nesten alle med uoppgitt
//     utdanning på nivåene, de fleste på grunnskole. Trolig fyller SSB inn utdanning (f.eks. for innvandrere)
//     som NUDB_BU ikke har. Ikke bekreftet. Bruk utdanningsfordelingen her som eget mål, ikke som SSBs tall.
// =====================================================================

require no.ssb.fdb:60 as fdb

// --- Populasjon: alle med NEET-verdi (bosatte 15–29 år, november 2024) ---
create-dataset unge
import fdb/ARBSTATUS_PERS_NEET 2024-11-16 as neet
import fdb/ARBSTATUS_PERS_STATUS8 2024-11-16 as status
import fdb/ARBSTATUS_PERS_ALDER 2024-11-16 as alder
import fdb/ARBSTATUS_PERS_KOMM_NR 2024-11-30 as kommune
import fdb/BEFOLKNING_KJOENN as kjonn
import fdb/BEFOLKNING_INVKAT as invkat
import fdb/NUDB_BU 2024-08-01 as utd

// --- Hva NEET-gruppen gjør (hovedstatus, de fire første sifrene) ---
generate status4 = substr(status, 1, 4)
generate neetstat = '0'
replace neetstat = '4' if neet == '1'
replace neetstat = '1' if neet == '1' & status4 == '2408'
replace neetstat = '2' if neet == '1' & (status4 == '4848' | status4 == '4864')
replace neetstat = '3' if neet == '1' & status4 == '7208'
define-labels neetstattxt '0' 'Ikke NEET' '1' 'NEET: registrert ledig' '2' 'NEET: AAP/uføretrygd' '3' 'NEET: ukjent status' '4' 'NEET: andre ytelser'
assign-labels neetstat neetstattxt

// --- Aldersgrupper ---
generate aldgr = '1'
replace aldgr = '2' if alder >= 20
replace aldgr = '3' if alder >= 25
define-labels aldgrtxt '1' '15–19 år' '2' '20–24 år' '3' '25–29 år'
assign-labels aldgr aldgrtxt

// --- Fylke (2024-inndelingen, som fylke.txt) ---
generate fylke = substr(kommune, 1, 2)
define-labels fylketxt '03' 'Oslo' '11' 'Rogaland' '15' 'Møre og Romsdal' '18' 'Nordland' '31' 'Østfold' '32' 'Akershus' '33' 'Buskerud' '34' 'Innlandet' '39' 'Vestfold' '40' 'Telemark' '42' 'Agder' '46' 'Vestland' '50' 'Trøndelag' '55' 'Troms' '56' 'Finnmark' '21' 'Svalbard' '99' 'Uoppgitt'
assign-labels fylke fylketxt

// --- Innvandrere og resten (som SSB 12423) ---
generate innv = '2'
replace innv = '1' if invkat == 'B'
define-labels innvtxt '1' 'Innvandrere' '2' 'Befolkningen utenom innvandrere'
assign-labels innv innvtxt

// --- Utdanningsnivå (som SSB 12423: fagskole regnes som videregående nivå) ---
generate utd1 = substr(utd, 1, 1)
generate utdniv = '9'
replace utdniv = '1' if utd1 == '1' | utd1 == '2'
replace utdniv = '2' if utd1 == '3' | utd1 == '4' | utd1 == '5'
replace utdniv = '3' if utd1 == '6' | utd1 == '7' | utd1 == '8'
define-labels utdnivtxt '1' 'Grunnskolenivå' '2' 'Videregående nivå (inkl. fagskole)' '3' 'Universitet og høgskole' '9' 'Uoppgitt eller ingen'
assign-labels utdniv utdnivtxt
