// =====================================================================
// Prioritert arbeidsstyrkestatus i åtte grupper
// Status:    testet i større skript (no.ssb.fdb:57, 2026-11)
// Variabler: ARBSTATUS_PERS_STATUS8 (status, 4. kvartal; 2008-11-16 til 2024-11-16)
//            https://microdata.no/discovery/variable/no.ssb.fdb/58/ARBSTATUS_PERS_STATUS8
// Lager:     arbstatus – '1' sysselsatt, '2' registrert ledig, '3' tiltak, '4' utdanning,
//            '5' AAP/uføre, '6' AFP/alderspensjon, '7' andre stønader, '8' ukjent/ikke bosatt
// Fallgruver:
//   - Koden har åtte sifre: de fire første er hovedstatus, de fire siste tilleggsstatus.
//     Grupperingen bruker bare hovedstatus (substr 1–4).
//   - Måledato er 16. november hvert år. Datoen må ligge innenfor dekningen.
//   - REGSYS_YRKSTAT gjelder bare til og med 2014.
// =====================================================================

require no.ssb.fdb:57 as fdb

create-dataset personer
import fdb/BEFOLKNING_KOMMNR_FORMELL 2025-01-01 as kommune
import fdb/ARBSTATUS_PERS_STATUS8 2024-11-16 as status

generate status4 = substr(status, 1, 4)
generate arbstatus = '8'
replace arbstatus = '1' if status4 == '0808' | status4 == '0816' | status4 == '0832' | status4 == '1608'
replace arbstatus = '2' if status4 == '2408'
replace arbstatus = '3' if status4 == '3208' | status4 == '3216' | status4 == '3240' | status4 == '4816'
replace arbstatus = '4' if status4 == '4008'
replace arbstatus = '5' if status4 == '4848' | status4 == '4864'
replace arbstatus = '6' if status4 == '5616' | status4 == '5632'
replace arbstatus = '7' if status4 == '3224' | status4 == '4808' | status4 == '5608' | status4 == '5624' | status4 == '5640' | status4 == '6408' | status4 == '6416' | status4 == '6424' | status4 == '6432'
define-labels arbstatustxt '1' 'Sysselsatt' '2' 'Registrert ledig' '3' 'Arbeidsmarkedstiltak' '4' 'Under utdanning' '5' 'AAP/uføretrygd' '6' 'AFP/alderspensjon' '7' 'Andre (dagpenger, sykepenger, stønader mv.)' '8' 'Ukjent / ikke bosatt'
assign-labels arbstatus arbstatustxt

textblock

«Ukjent / ikke bosatt» omfatter barn under 15 og personer som ikke var bosatt i november 2024.
endblock
