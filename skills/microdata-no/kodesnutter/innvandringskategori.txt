require no.ssb.fdb:59 as fdb

create-dataset personer
import fdb/BEFOLKNING_KOMMNR_FORMELL 2025-01-01 as kommune
import fdb/BEFOLKNING_INVKAT as invkat

generate innvkat = '3'
replace innvkat = '1' if invkat == 'B'
replace innvkat = '2' if invkat == 'C'
replace innvkat = '9' if sysmiss(invkat)
define-labels innvkattxt '1' 'Innvandrer' '2' 'Norskfødt med innvandrerforeldre' '3' 'Øvrig befolkning' '9' 'Ukjent'
assign-labels innvkat innvkattxt

textblock
## Bosatte etter innvandringskategori, 1.1.2025
endblock
tabulate innvkat, missing
